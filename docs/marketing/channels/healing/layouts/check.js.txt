 // 文字實際範圍（含溢出），與容器框取聯集，旋轉元素用外接框
 const t=rg.getBoundingClientRect();return {o:e.dataset.o,x:Math.min(r.left,t.width?t.left:r.left),y:Math.min(r.top,t.height?t.top:r.top),w:Math.max(r.right,t.width?t.right:r.right),h:Math.max(r.bottom,t.height?t.bottom:r.bottom)}});
